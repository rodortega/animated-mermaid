import { animateMermaid } from './animate-mermaid.js';

const ctl = await animateMermaid(document.querySelector('#diagram'), source, {
  theme: 'ocean',          // aurora, ember, mono, ocean, forest, sunset, neon, light, contrast
  nodeStyle: 'glass',      // gradient, glass, neon, solid, outline, soft
  corners: 'rounded',      // sharp, rounded, pill
  curve: 'basis',          // basis (smooth), linear (straight), step
  thickness: 'normal',     // thin, normal, thick
  glow: true,
  preset: 'cascade',       // build, cascade, fade, slide, typewriter, pulse, none
  particleStyle: 'comet',  // dot, comet, spark, arrow
  particles: 3, speed: 'normal', flow: 'off', loop: false,
  panZoom: true,           // ctrl/cmd + wheel, pinch, drag, keys + - 0
  onViewChange: ({ zoom, fullscreen }) => {},
});

ctl.zoomIn(); ctl.zoomOut(); ctl.setZoom(1.5); ctl.fit(); ctl.toggleFullscreen();
await ctl.set({ preset: 'pulse', theme: 'light', nodeStyle: 'soft' });
ctl.pause(); ctl.play(); ctl.replay();
ctl.setSpeed('fast'); ctl.setParticles(5); ctl.setFlow('slow');
